
`timescale 1ns / 1ps

module resize_coords (
    input  wire [15:0] in_w,
    input  wire [15:0] in_h,
    input  wire [15:0] scaled_w,
    input  wire [15:0] scaled_h,
    input  wire [15:0] pad_left,
    input  wire [15:0] pad_top,

    // Retained for compatibility with resize_top.
    // The new mapping uses actual scaled dimensions.
    input  wire [15:0] scale_num,
    input  wire [15:0] scale_den,

    input  wire [15:0] x_out,
    input  wire [15:0] y_out,

    output reg         pixel_valid,
    output reg [15:0]  x0,
    output reg [15:0]  x1,
    output reg [15:0]  y0,
    output reg [15:0]  y1,
    output reg [8:0]   a,
    output reg [8:0]   b
);

    reg [15:0] x_local;
    reg [15:0] y_local;

    // 16 x 16 multiplication, with extra width for Q8.
    reg [31:0] x_mult;
    reg [31:0] y_mult;

    reg [39:0] x_product;
    reg [39:0] y_product;
    reg [39:0] x_fixed;
    reg [39:0] y_fixed;

    reg [15:0] x_base;
    reg [15:0] y_base;

    always @(*) begin
        pixel_valid = 1'b0;

        x0 = 16'd0;
        x1 = 16'd0;
        y0 = 16'd0;
        y1 = 16'd0;
        a  = 9'd0;
        b  = 9'd0;

        x_local = 16'd0;
        y_local = 16'd0;

        x_mult = 32'd0;
        y_mult = 32'd0;

        x_product = 40'd0;
        y_product = 40'd0;
        x_fixed = 40'd0;
        y_fixed = 40'd0;

        x_base = 16'd0;
        y_base = 16'd0;

        if ((in_w != 0) &&
            (in_h != 0) &&
            (scaled_w != 0) &&
            (scaled_h != 0)) begin

            // Check if output coordinate lies inside the
            // resized image area, excluding padding.
            if (({1'b0, x_out} >= {1'b0, pad_left}) &&
                ({1'b0, x_out} <
                    ({1'b0, pad_left} + {1'b0, scaled_w})) &&
                ({1'b0, y_out} >= {1'b0, pad_top}) &&
                ({1'b0, y_out} <
                    ({1'b0, pad_top} + {1'b0, scaled_h}))) begin

                pixel_valid = 1'b1;

                x_local = x_out - pad_left;
                y_local = y_out - pad_top;

                // Map output pixel coordinates to source
                // coordinates in Q8 fixed-point format.
                x_mult = x_local * in_w;
                y_mult = y_local * in_h;

                x_product = {x_mult, 8'b0};
                y_product = {y_mult, 8'b0};

                // Rounded division by actual scaled dimensions.
                x_fixed =
                    (x_product + (scaled_w / 16'd2)) / scaled_w;

                y_fixed =
                    (y_product + (scaled_h / 16'd2)) / scaled_h;

                x_base = x_fixed[23:8];
                y_base = y_fixed[23:8];

                // Clamp horizontal neighbors to source bounds.
                if (x_base >= in_w)
                    x0 = in_w - 16'd1;
                else
                    x0 = x_base;

                if (x_base >= (in_w - 16'd1))
                    x1 = in_w - 16'd1;
                else
                    x1 = x_base + 16'd1;

                // Clamp vertical neighbors to source bounds.
                if (y_base >= in_h)
                    y0 = in_h - 16'd1;
                else
                    y0 = y_base;

                if (y_base >= (in_h - 16'd1))
                    y1 = in_h - 16'd1;
                else
                    y1 = y_base + 16'd1;

                // Fractional source coordinates in Q8.
                a = {1'b0, x_fixed[7:0]};
                b = {1'b0, y_fixed[7:0]};
            end
        end
    end

endmodule
